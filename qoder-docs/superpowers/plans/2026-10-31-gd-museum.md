**Goal:** 在 `canton-atlas` 里新增广东省博物馆（新馆）纯外观 1:1 近景（`gd-museum.html`）：约 113.5 m 见方、44.5 m 高的方形「回」字盒体，67.5 m 核心 + 四边各 23 m 悬挑，深灰铝板 + 红色凹龛 + 玻璃/格栅 + 锯齿顶边，约 40 m 中央天窗，西门台阶/草坡/「丝绸」景观，花城广场周边压缩城市；三档时段、7 视点、7 热点、4K、移动端；接入图鉴 `gd-museum` 深链与城市全景通用返回。

**Architecture:** 沿用 `memorial.*` 的单页近景架构：`gd-museum-textures.js` 程序化生成纹理，`gd-museum-model.js` 用 76×24 模数的合并墙面 + 洞口凹龛 + 屋顶网格建盒体与广场，`gd-museum-main.js` 负责渲染、运镜、视点、时段、热点、4K 与 `GdMuseumDebug`。盒体墙面/红内壁/玻璃/格栅全部合并或实例化，不依赖 `BufferGeometryUtils`；城市/树/格栅用实例化；不修改现有 9 个近景与城市 POC 的模型。

**Tech Stack:** Three.js（共享 `immersive/three.min.js`）、原生 JS/HTML/CSS、Node 24（内置 WebSocket/fetch）CDP 验收、Chrome 1440×900 / 390×844、Node 静态服务器 `.preview/atlas-server.mjs`。

## Global Constraints

- 坐标：1 场景单位 = 1 米；盒体中心 (0,0,0)；**+Z 南、-Z 北、+X 东、-X 西**；西广场 y=0。
- 盒体：x/z ∈ [-56.75, +56.75]；y ∈ [9, 44.5]；核心筒 x/z ∈ [-33.75, +33.75]；底层玻璃大堂 y ∈ [3, 9]；入口平台 y=3；中央天窗 40×40、y=38；屋顶 44.5。
- 盒体/核心/四边 23 m 悬挑/入口广场主要尺度 1:1；周边城市距离/体量压缩 0.7–0.75×；所有页面标注「周边压缩示意，非测绘」。
- 展厅/内部/文物/电梯/办公/安检流程不做；中央天窗只做屋顶玻璃，不做中庭内部。
- 立面洞口约 100–120 个（桌面）、1.5 m 模数、深度 0.6–2.4 m 为**照片估算/设计假定**；不写官方开孔率；不写「只有四个切口」。
- 表皮是**铝板（不写铜板）** + 玻璃 + 花岗石；红内壁是涂料/饰面，夜间红光为示意。
- 三档时段内部名固定 `day` / `dusk` / `night`；UI「晴日」「黄昏」「夜景」；按钮循环 day → dusk → night → day；`night` 视点强制夜景。
- 所有纹理程序化生成，不下载外部图片作为贴图；参考图只做内部观察，不进入站点资源。
- 性能目标：桌面 1440×900 固定视点 60fps；移动 390×844 固定视点 60fps；切换瞬态 ≥ 50fps；0 console error；无横向溢出。

- [ ] **Step 4: 纹理骨架**

创建 `gd-museum-textures.js`，使用 memorial 的 `canvasTexture` 模式：

```js
(function () {
  "use strict";
  function canvasTexture(draw, w, h, rx, ry) {
    var c = document.createElement("canvas");
    c.width = w; c.height = h;
    var ctx = c.getContext("2d");
    draw(ctx, w, h);
    var t = new THREE.CanvasTexture(c);
    t.encoding = THREE.sRGBEncoding;
    t.wrapS = t.wrapT = THREE.RepeatWrapping;
    if (rx || ry) t.repeat.set(rx || 1, ry || 1);
    t.needsUpdate = true;
    return t;
  }
  function build(renderer) {
    var metal = canvasTexture(function (ctx, w, h) {
      ctx.fillStyle = "#4b5053"; ctx.fillRect(0, 0, w, h);
      ctx.strokeStyle = "rgba(16,18,19,.55)"; ctx.lineWidth = 2;
      for (var i = 0; i <= w; i += w / 4) {
        ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i, h); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(w, i); ctx.stroke();
      }
      for (var k = 0; k < 1600; k++) {
        var a = Math.random() * 0.06;
        ctx.fillStyle = "rgba(255,255,255," + a + ")";
        ctx.fillRect(Math.random() * w, Math.random() * h, 2, 2);
      }
    }, 512, 512);
    // Task 3 补全：redPaint / glass / granite / paving / grass / leaf / palmLeaf / trunk / cityWindow / sign / roofGravel / glow
    return {
      metal: metal, metalRough: metal, redPaint: metal, glass: metal,
      granite: metal, paving: metal, grass: metal, leaf: metal,
      palmLeaf: metal, trunk: metal, cityWindow: metal,
      sign: null, roofGravel: metal, glow: null
    };
  }
  window.GdMuseumTextures = { build: build };
})();
```

function mulberry32(a) {
  return function () {
    a |= 0; a = a + 0x6D2B79F5 | 0;
    var t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}
function makeCuts(side, low) {
  var rand = mulberry32(0x9e3779b9 + side * 0x85ebca6b);
  var occ = new Uint8Array(COLS * ROWS);
  function ix(u, v) { return v * COLS + u; }
  function free(u, v, w, h) {
    if (u < 1 || v < 0 || u + w > COLS - 1 || v + h > ROWS) return false;
    for (var j = 0; j < h; j++) for (var i = 0; i < w; i++) if (occ[ix(u + i, v + j)]) return false;
    return true;
  }
  function mark(u, v, w, h, val) {
    for (var j = 0; j < h; j++) for (var i = 0; i < w; i++) occ[ix(u + i, v + j)] = val || 1;
  }
  var cuts = [];
  function put(u, v, w, h, type, depth, extra) {
    if (!free(u, v, w, h)) return false;
    cuts.push({ u: u, v: v, w: w, h: h, type: type, depth: depth, extra: extra || "" });
    mark(u, v, w, h, 1);
    return true;
  }
  // 西门 hero：入口上方底部缺口（30 m × 6 m）
  if (side === 0) {
    put(28, 0, 20, 4, "open", 2.4, "entrance");
    mark(36, 7, 17, 6, 2); // 字牌实板区（2 = 保留，不参与洞口）
  }
  // 每面 1 个竖槽 + 1 个 L 形（按面变化）
  var hero = [
    [[8, 6, 3, 12], [60, 4, 8, 3], [65, 4, 3, 8]],
    [[10, 3, 4, 10], [56, 12, 9, 3], [56, 15, 3, 6]],
    [[14, 8, 3, 9], [52, 6, 8, 4], [57, 6, 3, 8]],
    [[9, 5, 4, 8], [58, 10, 8, 3], [63, 13, 3, 7]]
  ][side] || [];
  hero.forEach(function (r) {
    var type = r[2] >= 4 && r[3] >= 8 ? "red" : "glass";
    put(r[0], r[1], r[2], r[3], type, type === "glass" ? 0.6 : 1.4, type === "glass" ? "grille" : "");
  });
  // 随机洞口：桌面 26 个/面，移动 20 个/面
  var target = low ? 20 : 26;
  var guard = 0;
  while (cuts.length < target && guard < 900) {
    guard++;
    var w = [2, 3, 4, 5, 6, 8, 10][Math.floor(rand() * 7)];
    var h = [2, 3, 4, 5, 6][Math.floor(rand() * 5)];
    var u = 1 + Math.floor(rand() * (COLS - w - 2));
    var v = 1 + Math.floor(rand() * (ROWS - h - 2));
    if (!free(u, v, w, h)) continue;
    var r = rand();
    var type = r < 0.40 ? "glass" : r < 0.85 ? "red" : "open";
    var depth = type === "glass" ? 0.6 : type === "red" ? 0.9 + rand() * 1.2 : 2.0 + rand() * 0.6;
    depth = Math.round(depth * 10) / 10;
    put(u, v, w, h, type, depth, type === "glass" && rand() < 0.7 ? "grille" : "");
  }
  // 顶部缺口：3–5 处，1–3 行深
  var n = low ? 3 : 5;
  for (var k = 0; k < n; k++) {
    var w2 = 2 + Math.floor(rand() * 3);
    var h2 = 1 + Math.floor(rand() * 3);
    var u2 = 2 + Math.floor(rand() * (COLS - w2 - 4));
    if (free(u2, ROWS - h2, w2, h2)) {
      cuts.push({ u: u2, v: ROWS - h2, w: w2, h: h2, type: "notch", depth: 0, extra: "" });
      mark(u2, ROWS - h2, w2, h2, 1);
    }
  }
  return { cuts: cuts, occ: occ };
}

```js
function faceBasis(side) {
  // origin(u=0,v=0) + uDir*MX*u + vDir*MY*v; normal 朝外
  if (side === 0) return { o: [-HALF, BOTTOM, -HALF], u: [0, 0, 1], v: [0, 1, 0], n: [-1, 0, 0] };
  if (side === 1) return { o: [-HALF, BOTTOM, HALF], u: [1, 0, 0], v: [0, 1, 0], n: [0, 0, 1] };
  if (side === 2) return { o: [HALF, BOTTOM, HALF], u: [0, 0, -1], v: [0, 1, 0], n: [1, 0, 0] };
  return { o: [HALF, BOTTOM, -HALF], u: [-1, 0, 0], v: [0, 1, 0], n: [0, 0, -1] };
}
function wp(b, u, v) {
  return [b.o[0] + b.u[0] * MX * u + b.v[0] * MY * v,
          b.o[1] + b.u[1] * MX * u + b.v[1] * MY * v,
          b.o[2] + b.u[2] * MX * u + b.v[2] * MY * v];
}
```

- [ ] **Step 4: 合并墙面与凹龛**

实现一个简单的 BufferGeometry 累加器（positions/normals/uvs/indices），不要依赖 `BufferGeometryUtils`：

```js
function GeoAcc() { this.p = []; this.n = []; this.uv = []; this.idx = []; }
GeoAcc.prototype.quad = function (a, b, c, d, n, uvs) {
  var base = this.p.length / 3;
  var pts = [a, b, c, d];
  for (var i = 0; i < 4; i++) {
    this.p.push(pts[i][0], pts[i][1], pts[i][2]);
    this.n.push(n[0], n[1], n[2]);
  }
  for (var k = 0; k < 8; k++) this.uv.push(uvs[k]);
  this.idx.push(base, base + 1, base + 2, base, base + 2, base + 3);
};
GeoAcc.prototype.build = function () {
  var g = new THREE.BufferGeometry();
  g.setAttribute("position", new THREE.Float32BufferAttribute(this.p, 3));
  g.setAttribute("normal", new THREE.Float32BufferAttribute(this.n, 3));
  g.setAttribute("uv", new THREE.Float32BufferAttribute(this.uv, 2));
  g.setIndex(this.idx);
  g.computeBoundingSphere();
  return g;
};
```

墙面：对每个 `(u,v)`，若 `occ[ix]===0`，生成一个朝外的 quad（角点按面基 + 法线）；UV 用 `u/4, v/4`（纹理 4×4 板）。红内壁：对每个 cut（非 notch），在洞口四边 + 背面生成 5 个朝内的 quad；玻璃窗背面另生成玻璃 quad（深度 `depth-0.03`）。示例：

```js
function buildFacades(textures, low) {
  var metalAcc = new GeoAcc(), redAcc = new GeoAcc(), glassAcc = new GeoAcc(), darkAcc = new GeoAcc();
  var stats = { cols: COLS, rows: ROWS, cuts: 0, glazed: 0, redRecess: 0, open: 0, topNotches: 0, drawCalls: 0, triangles: 0 };
  var allCuts = [];
  for (var side = 0; side < 4; side++) {
    var b = faceBasis(side);
    var made = makeCuts(side, !!low);
    var occ = made.occ, cuts = made.cuts;
    allCuts.push({ side: side, cuts: cuts });
    // 墙面：occ===1 才是洞口；occ===2 是字牌保留的实板
    for (var v = 0; v < ROWS; v++) for (var u = 0; u < COLS; u++) {
      var cell = occ[v * COLS + u];
      if (cell === 1) continue;
      var p0 = wp(b, u, v), p1 = wp(b, u + 1, v), p2 = wp(b, u + 1, v + 1), p3 = wp(b, u, v + 1);
      metalAcc.quad(p0, p1, p2, p3, b.n, [u / 4, v / 4, (u + 1) / 4, v / 4, (u + 1) / 4, (v + 1) / 4, u / 4, (v + 1) / 4]);
    }
    cuts.forEach(function (c) {
      if (c.type === "notch") { stats.topNotches++; return; }
      stats.cuts++;
      if (c.type === "glass") stats.glazed++;
      else if (c.type === "red") stats.redRecess++;
      else stats.open++;
      var depth = c.depth;
      var u0 = c.u, u1 = c.u + c.w, v0 = c.v, v1 = c.v + c.h;
      var inward = [-b.n[0], -b.n[1], -b.n[2]];
      function off(p, d) { return [p[0] + inward[0] * d, p[1] + inward[1] * d, p[2] + inward[2] * d]; }
      var a0 = wp(b, u0, v0), a1 = wp(b, u1, v0), a2 = wp(b, u1, v1), a3 = wp(b, u0, v1);
      var d0 = off(a0, depth), d1 = off(a1, depth), d2 = off(a2, depth), d3 = off(a3, depth);
      // 左/右/上/下侧壁（朝内）；v0==0 不建下壁，v1==ROWS 不建上壁
      redAcc.quad(d0, d1, a1, a0, [b.u[0], b.u[1], b.u[2]], [0,0, 1,0, 1,1, 0,1]);
      redAcc.quad(a3, a2, d2, d3, [-b.u[0], -b.u[1], -b.u[2]], [0,0, 1,0, 1,1, 0,1]);
      if (v0 > 0) redAcc.quad(a0, a3, d3, d0, [b.v[0], b.v[1], b.v[2]], [0,0, 1,0, 1,1, 0,1]);
      if (v1 < ROWS) redAcc.quad(d1, d2, a2, a1, [-b.v[0], -b.v[1], -b.v[2]], [0,0, 1,0, 1,1, 0,1]);
      // 背面
      var back = c.type === "glass" ? glassAcc : c.type === "red" ? redAcc : darkAcc;
      back.quad(d0, d1, d2, d3, b.n, [0,0, 1,0, 1,1, 0,1]);
    });
  }
  // 屋顶/核心/字牌由后续函数加入；此处先返回统计与全部洞口（供屋顶缺口使用）
  return { metalAcc: metalAcc, redAcc: redAcc, glassAcc: glassAcc, darkAcc: darkAcc, stats: stats, allCuts: allCuts };
}
```

function buildRibbon(points, width, material) {
  var curve = new THREE.CatmullRomCurve3(points);
  var seg = 80, pos = [], uv = [], idx = [];
  for (var i = 0; i <= seg; i++) {
    var t = i / seg, p = curve.getPoint(t), tan = curve.getTangent(t);
    var nx = -tan.z, nz = tan.x, len = Math.hypot(nx, nz) || 1;
    nx /= len; nz /= len;
    pos.push(p.x - nx * width/2, 0.06, p.z - nz * width/2);
    pos.push(p.x + nx * width/2, 0.06, p.z + nz * width/2);
    uv.push(0, t * 20, 1, t * 20);
    if (i < seg) { var a = i*2; idx.push(a, a+1, a+2, a+1, a+3, a+2); }
  }
  var g = new THREE.BufferGeometry();
  g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
  g.setIndex(idx); g.computeVertexNormals();
  return new THREE.Mesh(g, material);
}

function shot4K(opts) {
  opts = opts || {};
  return new Promise(function (resolve) {
    var oldW = width, oldH = height, oldRatio = renderer.getPixelRatio(), oldAspect = camera.aspect;
    function restore() {
      renderer.setPixelRatio(oldRatio); renderer.setSize(oldW, oldH, false);
      camera.aspect = oldAspect; camera.updateProjectionMatrix();
    }
    try {
      renderer.setPixelRatio(1);
      renderer.setSize(3840, 2160, false);
      camera.aspect = 3840 / 2160; camera.updateProjectionMatrix();
      renderer.render(scene, camera);
      canvas.toBlob(function (blob) {
        if (!blob) { restore(); resolve({ error: "toBlob null" }); return; }
        if (opts.download) {
          var a = document.createElement("a"), url = URL.createObjectURL(blob);
          a.href = url; a.download = "gd-museum-4k.png"; a.click();
          setTimeout(function () { URL.revokeObjectURL(url); }, 1000);
        }
        restore();
        resolve({ width: 3840, height: 2160, bytes: blob.size, type: blob.type });
      }, "image/png");
    } catch (e) {
      restore();
      resolve({ error: e.message || "4K failed" });
    }
  });
}

## 风险与回退

- **立面图案不似**：先用生成器 + 3–5 个 hero 洞（入口、竖槽、L 形、顶部缺口）定调；截图后调宽高/密度；接受"神似而非逐洞"。
- **红内壁过曝/塑料**：降低 emissive 到 1.4–1.8，增加暗部红/黑；用 ACES；不加 bloom。
- **玻璃排序**：不透明深色玻璃 + emissive 背面；避免大面积透明；必要时 `depthWrite:false` + renderOrder。
- **阴影 acne**：bias/normalBias 调整；薄洞口用暗部/顶点色补；移动/夜间关阴影。
- **顶部缺口没做**：Task 3 Step 5 必须补 `inNotch`；否则屋顶轮廓不对。
- **城市压缩被误读**：页面/About 明确「周边压缩，非测绘」；塔 0.75×；不写真实距离。
- **移动端超预算**：先减城市块/树/格栅/东/北/南洞口；不降西门/盒体/天窗。
- **接入残差**：城市通用返回当前落到 `chenjiaci`；记录，不擅自改父页。
- **命名**：广东省博物馆/月光宝盒；不写广州博物馆/镇海楼。
- **版权**：纹理全程序化；参考图只观察；不下载版权贴图；字牌只写馆名。
